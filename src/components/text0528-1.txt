import { describe, it, expect, vi, beforeEach, afterEach } from "vitest";
import { render, screen, fireEvent, waitFor, act } from "@testing-library/react";
import userEvent from "@testing-library/user-event";
import "@testing-library/jest-dom";

// ── テストデータ ──────────────────────────────────────────────

const TEST_INFO_TYPES = [
  {
    id: "00000000-0100-0000-0000-000000000000",
    nameJp: "情報種別AAA",
    nameEn: "InfoTypeAAA",
  },
  {
    id: "00000000-0200-0000-0000-000000000000",
    nameJp: "情報種別BBB",
    nameEn: "InfoTypeBBB",
  },
  {
    id: "00000000-0300-0000-0000-000000000000",
    nameJp: "情報種別CCC",
    nameEn: "InfoTypeCCC",
  },
];

const TEST_STATUSES = [
  { searchId: "100", searchNameJp: "一時保存", searchNameEn: "Draft" },
  { searchId: "200", searchNameJp: "公開済", searchNameEn: "published" },
];

const TEST_SLS = [
  { id: 1, name: "1" },
  { id: 2, name: "2" },
  { id: 3, name: "3" },
];

const TEST_ECTS = [
  { id: 1, nameJp: "非該当", nameEn: "none" },
  { id: 2, nameJp: "6項(3)/Plasma TBC", nameEn: "EN/Plasma TBC" },
  { id: 3, nameJp: "13項(3)/R1HGP", nameEn: "EN/13_r1hgp" },
];

// ── モジュールモック ──────────────────────────────────────────

vi.mock("../src/utils/apiUtils", () => ({
  postEDMS: vi.fn(),
}));

vi.mock("../src/constants/messages", () => ({
  MESSAGES: {
    DISPLAY_DATE_ERROR: "表示日付の開始日は終了日より前の日付を入力してください。",
    UPDATE_DATE_ERROR: "更新日時の開始日時は終了日時より前の日時を入力してください。",
  },
}));

vi.mock("../src/components/messageSnackbar", () => ({
  MessageSnackbar: ({ open, message, severity, onClose }: any) =>
    open ? (
      <div data-testid="snackbar" data-severity={severity}>
        {message}
        <button onClick={onClose}>close</button>
      </div>
    ) : null,
}));

vi.mock("../src/components/loading", () => ({
  Loading: ({ loading }: any) =>
    loading ? <div data-testid="loading" /> : null,
}));

vi.mock("@mui/material/Collapse", () => ({
  default: ({ in: inProp, children }: any) =>
    inProp ? <div data-testid="collapse-open">{children}</div> : null,
}));

vi.mock("@mui/material/FormControl", () => ({
  default: ({ children }: any) => <div>{children}</div>,
}));

vi.mock("@mui/material/InputLabel", () => ({
  default: ({ children, htmlFor }: any) => (
    <label htmlFor={htmlFor}>{children}</label>
  ),
}));

vi.mock("@mui/material/Select", () => ({
  default: ({ children, onChange, value, id }: any) => (
    <select id={id} onChange={onChange} value={value ?? ""}>
      {children}
    </select>
  ),
}));

vi.mock("@mui/material/MenuItem", () => ({
  default: ({ children, value }: any) => (
    <option value={value}>{children}</option>
  ),
}));

import SearchPage from "../src/search/searchPage";
import { postEDMS } from "../src/utils/apiUtils";

// ── モックレスポンス設定ヘルパー ──────────────────────────────

const mockAllApis = () => {
  (postEDMS as ReturnType<typeof vi.fn>)
    .mockResolvedValueOnce({ infoTypes: TEST_INFO_TYPES })
    .mockResolvedValueOnce({ infoEditionStatusConditions: TEST_STATUSES })
    .mockResolvedValueOnce({ sls: TEST_SLS })
    .mockResolvedValueOnce({ ects: TEST_ECTS });
};

const mockAllApisEmpty = () => {
  (postEDMS as ReturnType<typeof vi.fn>)
    .mockResolvedValueOnce({ infoTypes: [] })
    .mockResolvedValueOnce({ infoEditionStatusConditions: [] })
    .mockResolvedValueOnce({ sls: [] })
    .mockResolvedValueOnce({ ects: [] });
};

// ── テストスイート ────────────────────────────────────────────

describe("SearchPage", () => {
  beforeEach(() => {
    vi.clearAllMocks();
  });

  afterEach(() => {
    document.body.classList.remove("centered");
  });

  // ── 初期レンダリング ────────────────────────────────────────

  it("初期表示時に必要な要素が表示されること", async () => {
    mockAllApisEmpty();

    render(<SearchPage />);

    expect(screen.getByText("EDMS | 検索・一覧")).toBeInTheDocument();
    expect(screen.getByRole("button", { name: "クリア" })).toBeInTheDocument();
    expect(screen.getByRole("button", { name: "検索" })).toBeInTheDocument();
    expect(screen.queryByTestId("loading")).not.toBeInTheDocument();
    expect(screen.queryByTestId("snackbar")).not.toBeInTheDocument();
  });

  // ── useEffect: body classList / document.title ──────────────

  it("マウント時に body へ centered クラスが付与され document.title が設定されること", () => {
    mockAllApisEmpty();

    render(<SearchPage />);

    expect(document.body.classList.contains("centered")).toBe(true);
    expect(document.title).toBe("EDMS | 検索・一覧");
  });

  it("アンマウント時に body から centered クラスが除去されること", () => {
    mockAllApisEmpty();

    const { unmount } = render(<SearchPage />);
    expect(document.body.classList.contains("centered")).toBe(true);

    unmount();

    expect(document.body.classList.contains("centered")).toBe(false);
  });

  // ── 初期データ取得: 正常系 ──────────────────────────────────

  it("4つの API エンドポイントが呼ばれること", async () => {
    mockAllApisEmpty();

    render(<SearchPage />);

    await waitFor(() => expect(postEDMS).toHaveBeenCalledTimes(4));
  });

  it("データ取得中は Loading が表示され完了後に非表示になること", async () => {
    (postEDMS as ReturnType<typeof vi.fn>).mockImplementation(
      () => new Promise((resolve) => setTimeout(() => resolve({
        infoTypes: TEST_INFO_TYPES,
        infoEditionStatusConditions: TEST_STATUSES,
        sls: TEST_SLS,
        ects: TEST_ECTS,
      }), 100))
    );

    render(<SearchPage />);

    expect(screen.getByTestId("loading")).toBeInTheDocument();
    await waitFor(() => expect(screen.queryByTestId("loading")).not.toBeInTheDocument(), { timeout: 500 });
  });

  // ── 初期データ取得: 異常系 ──────────────────────────────────

  it("API エラー時に error の Snackbar が表示され Loading が非表示になること", async () => {
    (postEDMS as ReturnType<typeof vi.fn>).mockRejectedValue(new Error("Network Error"));

    render(<SearchPage />);

    await waitFor(() => expect(screen.getByTestId("snackbar")).toBeInTheDocument());
    expect(screen.getByTestId("snackbar")).toHaveAttribute("data-severity", "error");
    expect(screen.getByText("Network Error")).toBeInTheDocument();
    expect(screen.queryByTestId("loading")).not.toBeInTheDocument();
  });

  it("Snackbar の閉じるボタンで非表示になること", async () => {
    (postEDMS as ReturnType<typeof vi.fn>).mockRejectedValue(new Error("err"));

    render(<SearchPage />);

    await waitFor(() => expect(screen.getByTestId("snackbar")).toBeInTheDocument());
    fireEvent.click(screen.getByRole("button", { name: "close" }));
    await waitFor(() => expect(screen.queryByTestId("snackbar")).not.toBeInTheDocument());
  });

  // ── 検索条件の折りたたみ ────────────────────────────────────

  it("セクションヘッダをクリックするたびに折りたたみ状態が切り替わること", async () => {
    mockAllApisEmpty();

    render(<SearchPage />);

    await waitFor(() => expect(screen.getByTestId("collapse-open")).toBeInTheDocument());

    const header = screen.getByText("検索条件").closest("[class*='section-head']")!;

    fireEvent.click(header);
    await waitFor(() => expect(screen.queryByTestId("collapse-open")).not.toBeInTheDocument());

    fireEvent.click(header);
    await waitFor(() => expect(screen.getByTestId("collapse-open")).toBeInTheDocument());
  });

  // ── フォーム入力 ────────────────────────────────────────────

  it("文書番号とタイトルが変更できること", async () => {
    mockAllApisEmpty();

    render(<SearchPage />);

    await waitFor(() => expect(screen.getByPlaceholderText(/AQGT26/)).toBeInTheDocument());

    const docInput = screen.getByPlaceholderText(/AQGT26/);
    expect(docInput).toHaveValue("");
    await userEvent.type(docInput, "DOC-001");
    await waitFor(() => expect(docInput).toHaveValue("DOC-001"));

    const titleInput = screen.getByPlaceholderText(/水平展開/);
    expect(titleInput).toHaveValue("");
    await userEvent.type(titleInput, "テストタイトル");
    await waitFor(() => expect(titleInput).toHaveValue("テストタイトル"));
  });

  it("表示日付 From / To が変更できること", async () => {
    mockAllApisEmpty();

    render(<SearchPage />);

    await waitFor(() => expect(postEDMS).toHaveBeenCalledTimes(4));
    const dateInputs = screen.getAllByDisplayValue("").filter((el) => (el as HTMLInputElement).type === "date");

    expect(dateInputs[0]).toHaveValue("");
    fireEvent.change(dateInputs[0], { target: { value: "2024-01-01" } });
    await waitFor(() => expect(dateInputs[0]).toHaveValue("2024-01-01"));

    expect(dateInputs[1]).toHaveValue("");
    fireEvent.change(dateInputs[1], { target: { value: "2024-12-31" } });
    await waitFor(() => expect(dateInputs[1]).toHaveValue("2024-12-31"));
  });

  it("更新日時 From / To が変更できること", async () => {
    mockAllApisEmpty();

    render(<SearchPage />);

    await waitFor(() => expect(postEDMS).toHaveBeenCalledTimes(4));
    const dtInputs = screen.getAllByDisplayValue("").filter((el) => (el as HTMLInputElement).type === "datetime-local");

    expect(dtInputs[0]).toHaveValue("");
    fireEvent.change(dtInputs[0], { target: { value: "2024-01-01T00:00" } });
    await waitFor(() => expect(dtInputs[0]).toHaveValue("2024-01-01T00:00"));

    expect(dtInputs[1]).toHaveValue("");
    fireEvent.change(dtInputs[1], { target: { value: "2024-12-31T23:59" } });
    await waitFor(() => expect(dtInputs[1]).toHaveValue("2024-12-31T23:59"));
  });

  it("過去履歴チェックボックスが変更できること", async () => {
    mockAllApisEmpty();

    render(<SearchPage />);

    await waitFor(() => expect(screen.getByLabelText("過去履歴を表示する")).toBeInTheDocument());
    const checkbox = screen.getByLabelText("過去履歴を表示する");
    expect(checkbox).not.toBeChecked();

    fireEvent.click(checkbox);

    await waitFor(() => expect(checkbox).toBeChecked());
  });

  it("各 Select が変更できること", async () => {
    mockAllApis();

    render(<SearchPage />);

    await waitFor(() => expect(postEDMS).toHaveBeenCalledTimes(4));

    const infoType = screen.getByRole("combobox", { name: /情報種別/ });
    expect(infoType).toHaveValue("");
    fireEvent.change(infoType, { target: { value: TEST_INFO_TYPES[0].id } });
    await waitFor(() => expect(infoType).toHaveValue(TEST_INFO_TYPES[0].id));

    const status = screen.getByRole("combobox", { name: /ステータス/ });
    expect(status).toHaveValue("");
    fireEvent.change(status, { target: { value: TEST_STATUSES[0].searchId } });
    await waitFor(() => expect(status).toHaveValue(TEST_STATUSES[0].searchId));

    const sl = screen.getByRole("combobox", { name: /^SL$/ });
    expect(sl).toHaveValue("");
    fireEvent.change(sl, { target: { value: String(TEST_SLS[0].id) } });
    await waitFor(() => expect(sl).toHaveValue(String(TEST_SLS[0].id)));

    const ect = screen.getByRole("combobox", { name: /^ECT$/ });
    expect(ect).toHaveValue("");
    fireEvent.change(ect, { target: { value: String(TEST_ECTS[0].id) } });
    await waitFor(() => expect(ect).toHaveValue(String(TEST_ECTS[0].id)));

    const ear99 = screen.getByRole("combobox", { name: /EAR99/ });
    expect(ear99).toHaveValue("");
    fireEvent.change(ear99, { target: { value: "1" } });
    await waitFor(() => expect(ear99).toHaveValue("1"));
  });

  // ── クリアボタン ────────────────────────────────────────────

  it("クリア押下でテキスト入力がリセットされること", async () => {
    mockAllApisEmpty();

    render(<SearchPage />);

    await waitFor(() => expect(screen.getByPlaceholderText(/AQGT26/)).toBeInTheDocument());

    const docInput = screen.getByPlaceholderText(/AQGT26/);
    await userEvent.type(docInput, "DOC-TEST");
    await waitFor(() => expect(docInput).toHaveValue("DOC-TEST"));

    const titleInput = screen.getByPlaceholderText(/水平展開/);
    await userEvent.type(titleInput, "テスト");
    await waitFor(() => expect(titleInput).toHaveValue("テスト"));

    fireEvent.click(screen.getByRole("button", { name: "クリア" }));

    await waitFor(() => {
      expect(docInput).toHaveValue("");
      expect(titleInput).toHaveValue("");
    });
  });

  it("クリア押下で各 Select がリセットされること", async () => {
    mockAllApis();

    render(<SearchPage />);

    await waitFor(() => expect(postEDMS).toHaveBeenCalledTimes(4));

    const infoType = screen.getByRole("combobox", { name: /情報種別/ });
    const status = screen.getByRole("combobox", { name: /ステータス/ });
    const sl = screen.getByRole("combobox", { name: /^SL$/ });
    const ect = screen.getByRole("combobox", { name: /^ECT$/ });
    const ear99 = screen.getByRole("combobox", { name: /EAR99/ });

    fireEvent.change(infoType, { target: { value: TEST_INFO_TYPES[0].id } });
    fireEvent.change(status, { target: { value: TEST_STATUSES[0].searchId } });
    fireEvent.change(sl, { target: { value: String(TEST_SLS[0].id) } });
    fireEvent.change(ect, { target: { value: String(TEST_ECTS[0].id) } });
    fireEvent.change(ear99, { target: { value: "1" } });

    await waitFor(() => {
      expect(infoType).toHaveValue(TEST_INFO_TYPES[0].id);
      expect(status).toHaveValue(TEST_STATUSES[0].searchId);
      expect(sl).toHaveValue(String(TEST_SLS[0].id));
      expect(ect).toHaveValue(String(TEST_ECTS[0].id));
      expect(ear99).toHaveValue("1");
    });

    fireEvent.click(screen.getByRole("button", { name: "クリア" }));

    await waitFor(() => {
      expect(infoType).toHaveValue("");
      expect(status).toHaveValue("");
      expect(sl).toHaveValue("");
      expect(ect).toHaveValue("");
      expect(ear99).toHaveValue("");
    });
  });

  it("クリア押下で過去履歴チェックボックスがリセットされること", async () => {
    mockAllApisEmpty();

    render(<SearchPage />);

    await waitFor(() => expect(screen.getByLabelText("過去履歴を表示する")).toBeInTheDocument());
    const checkbox = screen.getByLabelText("過去履歴を表示する");
    fireEvent.click(checkbox);
    await waitFor(() => expect(checkbox).toBeChecked());

    fireEvent.click(screen.getByRole("button", { name: "クリア" }));

    await waitFor(() => expect(checkbox).not.toBeChecked());
  });

  // ── 検索: 表示日付バリデーション ───────────────────────────

  it("表示日付 From が To より後の場合にエラーが表示されること", async () => {
    mockAllApisEmpty();

    render(<SearchPage />);

    await waitFor(() => expect(postEDMS).toHaveBeenCalledTimes(4));
    const dateInputs = screen.getAllByDisplayValue("").filter((el) => (el as HTMLInputElement).type === "date");
    fireEvent.change(dateInputs[0], { target: { value: "2024-12-31" } });
    fireEvent.change(dateInputs[1], { target: { value: "2024-01-01" } });

    fireEvent.click(screen.getByRole("button", { name: "検索" }));

    await waitFor(() => expect(screen.getByTestId("snackbar")).toBeInTheDocument());
    expect(screen.getByTestId("snackbar")).toHaveAttribute("data-severity", "error");
    expect(screen.getByText(/表示日付の開始日は終了日より前/)).toBeInTheDocument();
  });

  it("表示日付 From と To が同じ / From のみ / To のみの場合にエラーが表示されないこと", async () => {
    // From === To
    mockAllApisEmpty();
    const { unmount } = render(<SearchPage />);
    await waitFor(() => expect(postEDMS).toHaveBeenCalledTimes(4));
    let dateInputs = screen.getAllByDisplayValue("").filter((el) => (el as HTMLInputElement).type === "date");
    fireEvent.change(dateInputs[0], { target: { value: "2024-06-01" } });
    fireEvent.change(dateInputs[1], { target: { value: "2024-06-01" } });
    fireEvent.click(screen.getByRole("button", { name: "検索" }));
    await waitFor(() => expect(screen.queryByText(/表示日付の開始日は終了日より前/)).not.toBeInTheDocument());
    unmount();

    // From のみ
    vi.clearAllMocks();
    mockAllApisEmpty();
    const { unmount: unmount2 } = render(<SearchPage />);
    await waitFor(() => expect(postEDMS).toHaveBeenCalledTimes(4));
    dateInputs = screen.getAllByDisplayValue("").filter((el) => (el as HTMLInputElement).type === "date");
    fireEvent.change(dateInputs[0], { target: { value: "2024-12-31" } });
    fireEvent.click(screen.getByRole("button", { name: "検索" }));
    await waitFor(() => expect(screen.queryByText(/表示日付の開始日は終了日より前/)).not.toBeInTheDocument());
    unmount2();

    // To のみ
    vi.clearAllMocks();
    mockAllApisEmpty();
    render(<SearchPage />);
    await waitFor(() => expect(postEDMS).toHaveBeenCalledTimes(4));
    dateInputs = screen.getAllByDisplayValue("").filter((el) => (el as HTMLInputElement).type === "date");
    fireEvent.change(dateInputs[1], { target: { value: "2024-01-01" } });
    fireEvent.click(screen.getByRole("button", { name: "検索" }));
    await waitFor(() => expect(screen.queryByText(/表示日付の開始日は終了日より前/)).not.toBeInTheDocument());
  });

  // ── 検索: 更新日時バリデーション ───────────────────────────

  it("更新日時 From が To より後の場合にエラーが表示されること", async () => {
    mockAllApisEmpty();

    render(<SearchPage />);

    await waitFor(() => expect(postEDMS).toHaveBeenCalledTimes(4));
    const dtInputs = screen.getAllByDisplayValue("").filter((el) => (el as HTMLInputElement).type === "datetime-local");
    fireEvent.change(dtInputs[0], { target: { value: "2024-12-31T23:59" } });
    fireEvent.change(dtInputs[1], { target: { value: "2024-01-01T00:00" } });

    fireEvent.click(screen.getByRole("button", { name: "検索" }));

    await waitFor(() => expect(screen.getByTestId("snackbar")).toBeInTheDocument());
    expect(screen.getByTestId("snackbar")).toHaveAttribute("data-severity", "error");
    expect(screen.getByText(/更新日時の開始日時は終了日時より前/)).toBeInTheDocument();
  });

  it("更新日時 From と To が同じ / From のみ / To のみの場合にエラーが表示されないこと", async () => {
    // From === To
    mockAllApisEmpty();
    const { unmount } = render(<SearchPage />);
    await waitFor(() => expect(postEDMS).toHaveBeenCalledTimes(4));
    let dtInputs = screen.getAllByDisplayValue("").filter((el) => (el as HTMLInputElement).type === "datetime-local");
    fireEvent.change(dtInputs[0], { target: { value: "2024-06-01T12:00" } });
    fireEvent.change(dtInputs[1], { target: { value: "2024-06-01T12:00" } });
    fireEvent.click(screen.getByRole("button", { name: "検索" }));
    await waitFor(() => expect(screen.queryByText(/更新日時の開始日時は終了日時より前/)).not.toBeInTheDocument());
    unmount();

    // From のみ
    vi.clearAllMocks();
    mockAllApisEmpty();
    const { unmount: unmount2 } = render(<SearchPage />);
    await waitFor(() => expect(postEDMS).toHaveBeenCalledTimes(4));
    dtInputs = screen.getAllByDisplayValue("").filter((el) => (el as HTMLInputElement).type === "datetime-local");
    fireEvent.change(dtInputs[0], { target: { value: "2024-12-31T23:59" } });
    fireEvent.click(screen.getByRole("button", { name: "検索" }));
    await waitFor(() => expect(screen.queryByText(/更新日時の開始日時は終了日時より前/)).not.toBeInTheDocument());
    unmount2();

    // To のみ
    vi.clearAllMocks();
    mockAllApisEmpty();
    render(<SearchPage />);
    await waitFor(() => expect(postEDMS).toHaveBeenCalledTimes(4));
    dtInputs = screen.getAllByDisplayValue("").filter((el) => (el as HTMLInputElement).type === "datetime-local");
    fireEvent.change(dtInputs[1], { target: { value: "2024-01-01T00:00" } });
    fireEvent.click(screen.getByRole("button", { name: "検索" }));
    await waitFor(() => expect(screen.queryByText(/更新日時の開始日時は終了日時より前/)).not.toBeInTheDocument());
  });

  // ── 検索: 正常系 ────────────────────────────────────────────

  it("正常な条件で検索押下してもエラーが表示されないこと", async () => {
    mockAllApisEmpty();

    render(<SearchPage />);

    await waitFor(() => expect(postEDMS).toHaveBeenCalledTimes(4));

    const dateInputs = screen.getAllByDisplayValue("").filter((el) => (el as HTMLInputElement).type === "date");
    fireEvent.change(dateInputs[0], { target: { value: "2024-01-01" } });
    fireEvent.change(dateInputs[1], { target: { value: "2024-12-31" } });

    const dtInputs = screen.getAllByDisplayValue("").filter((el) => (el as HTMLInputElement).type === "datetime-local");
    fireEvent.change(dtInputs[0], { target: { value: "2024-01-01T00:00" } });
    fireEvent.change(dtInputs[1], { target: { value: "2024-12-31T23:59" } });

    await act(async () => {
      fireEvent.click(screen.getByRole("button", { name: "検索" }));
    });

    expect(screen.queryByTestId("snackbar")).not.toBeInTheDocument();
  });

  // ── アクセシビリティ ────────────────────────────────────────

  it("必要な aria-label と id が付与されていること", () => {
    mockAllApisEmpty();

    render(<SearchPage />);

    expect(screen.getByLabelText("画面タイトル")).toBeInTheDocument();
    expect(document.getElementById("clearBtn")).toBeInTheDocument();
    expect(document.getElementById("searchBtn")).toBeInTheDocument();
  });
});
