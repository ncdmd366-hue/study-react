USE [EDMS]
GO

DECLARE @file_path VARCHAR(255) = 'test'
DECLARE @file_name VARCHAR(255) = 'test.txt'

INSERT INTO [dbo].[T_FILE] (
    path, name, physical_del_flg, del_flg,
    created_by, created_at, updated_by, updated_at
) VALUES (
    @file_path, @file_name, 0, 0,
    'system', GETDATE(), 'system', GETDATE()
);

-- ★ このIDを次のUPDATEの @t_file_id に設定すること
SELECT SCOPE_IDENTITY() AS t_file_id;


USE [EDMS]
GO

-- ============================================================
-- ★ UPDATEパラメータ (ここだけ変更すればOK)
-- ============================================================
DECLARE @target_file_id UNIQUEIDENTIFIER = 'ここに対象のUUID' -- ②のSELECTで確認したid
DECLARE @t_file_id      INT              = 1                  -- ②で取得したt_file_id
-- ============================================================

UPDATE [dbo].[T_INFO_EDITION_FILE]
SET
    internal_file_flg = 1,
    t_file_id         = @t_file_id,
    external_url      = NULL
WHERE id = @target_file_id;

-- 確認
SELECT id, display_file_name, internal_file_flg, t_file_id
FROM [dbo].[T_INFO_EDITION_FILE]
WHERE id = @target_file_id;
