コントローラー

/**
 * 詳細取得：情報詳細取得
 * @returns InfoDetailResponseDto
 */
@ApiOperation({
  summary: '詳細取得：情報詳細取得',
})
@ApiResponse({
  status: 200,
  description: 'OK',
  type: InfoDetailResponseDto,
})
@ApiBadRequestResponse({
  description: 'Bad Request',
  type: BadRequestResponseDto,
})
@ApiInternalServerErrorResponse({
  description: 'Internal server error',
  type: InternalServerErrorResponseDto,
})
@Post('detail/info')
@HttpCode(200)
async infoDetail(
  @Body() infoDetailRequestDto: InfoDetailRequestDto,
) {
  //Validation Check
  this.docMngSearchService.validateInfoDetailRequest(
    infoDetailRequestDto,
  );
  //検索
  const detail = await this.docMngSearchService.findInfoDetail(
    infoDetailRequestDto,
  );
  //ResponseDtoに変換して返却
  return this.docMngSearchService.convertInfoDetailResponseDto(
    detail,
  );
}
