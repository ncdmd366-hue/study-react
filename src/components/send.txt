it("エラー時にSnackbarが表示され、閉じるボタンで閉じられること", async () => {
  const { postEDMS } = await import("../src/utils/apiUtils")
  vi.mocked(postEDMS).mockRejectedValueOnce(
    new Error("APIエラー")
  )

  render(<OperationFileTable />)

  // Snackbarが開いていることを確認
  await waitFor(() => {
    expect(screen.getByTestId("snackbar-open").textContent).toBe("true")
    expect(screen.getByTestId("snackbar-severity").textContent).toBe("error")
  })

  // 閉じるボタンをクリック
  const user = userEvent.setup()
  await user.click(screen.getByTestId("snackbar-close"))

  // Snackbarが閉じられることを確認
  await waitFor(() => {
    expect(screen.getByTestId("snackbar-open").textContent).toBe("false")
  })
})
