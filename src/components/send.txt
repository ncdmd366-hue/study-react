import { render, screen } from "@testing-library/react";
import { describe, it, expect } from "vitest";
import InfoDashboard from "./InfoDashboard";

describe("InfoDashboard", () => {
  it("テーブルラベルが表示される", () => {
    render(<InfoDashboard />);
    expect(screen.getByText("DATA_SOURCE（情報源）の列の項目")).toBeInTheDocument();
  });

  it("テーブルの全行が表示される", () => {
    render(<InfoDashboard />);
    expect(screen.getByText("アベイラビリティ会議時に登録されている運転情報より作成（※）")).toBeInTheDocument();
    expect(screen.getByText("Cloudから取得できる運転データより作成")).toBeInTheDocument();
    expect(screen.getByText(/から取得できる運転データより作成/)).toBeInTheDocument();
    expect(screen.getByText("上記の情報源では入手できないデータを社内のヒアリングで入手")).toBeInTheDocument();
  });

  it("社内ヒアリングのキーが表示される", () => {
    render(<InfoDashboard />);
    expect(screen.getByText("社内ヒアリング")).toBeInTheDocument();
  });
});
